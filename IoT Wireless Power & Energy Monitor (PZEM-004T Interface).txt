/**
 * @file PZEM_Energy_Monitor.ino
 * @brief IoT AC Power Telemetry Node using ESP32 and PZEM-004T
 * 
 * An internet-of-things embedded energy telemetry project programmed in Arduino IDE 
 * to capture and monitor real-time AC power consumption metrics.
 */

#include <PZEM004Tv30.h>

// Initialize PZEM-004T on ESP32 Hardware Serial 2 (RX = Pin 16, TX = Pin 17)
PZEM004Tv30 pzem(Serial2, 16, 17);

void setup() {
  // Start standard serial monitor for debugging and telemetry output
  Serial.begin(115200);
  delay(1000);
  
  Serial.println("Initializing PZEM-004T AC Power Monitor Node...");
}

void loop() {
  // Read real-time electrical parameters from the sensor module
  float voltage = pzem.voltage();
  float current = pzem.current();
  float power = pzem.power();
  float energy = pzem.energy();
  float frequency = pzem.frequency();
  float pf = pzem.pf();

  // Validate sensor communication status
  if (isnan(voltage)) {
    Serial.println("Error: Failed to read data from PZEM sensor module!");
  } else {
    // Print formatted telemetry stream to Serial Monitor
    Serial.println("================================");
    Serial.println("    AC POWER TELEMETRY DATA     ");
    Serial.println("================================");
    Serial.print("RMS Voltage:   "); Serial.print(voltage);   Serial.println(" V");
    Serial.print("RMS Current:   "); Serial.print(current);   Serial.println(" A");
    Serial.print("Active Power:  "); Serial.print(power);     Serial.println(" W");
    Serial.print("Energy Usage:  "); Serial.print(energy);    Serial.println(" kWh");
    Serial.print("Frequency:     "); Serial.print(frequency); Serial.println(" Hz");
    Serial.print("Power Factor:  "); Serial.println(pf);
  }

  Serial.println("--------------------------------\n");
  
  // Update interval set to 2 seconds
  delay(2000);
}
